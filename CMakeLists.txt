cmake_minimum_required(VERSION 3.16)
project(MustardTube VERSION 0.9.11 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
pkg_check_modules(SDL2_TTF REQUIRED IMPORTED_TARGET SDL2_ttf)
pkg_check_modules(SDL2_IMAGE REQUIRED IMPORTED_TARGET SDL2_image)
add_executable(mustardtube src/main.c)
target_link_libraries(mustardtube PRIVATE PkgConfig::SDL2 PkgConfig::SDL2_TTF PkgConfig::SDL2_IMAGE m)
target_compile_options(mustardtube PRIVATE -Wall -Wextra -Wpedantic)
if(CMAKE_BUILD_TYPE STREQUAL "Debug")
  target_compile_definitions(mustardtube PRIVATE MT_DEBUG=1)
endif()
install(TARGETS mustardtube RUNTIME DESTINATION bin)
install(DIRECTORY assets/ DESTINATION share/mustardtube)
